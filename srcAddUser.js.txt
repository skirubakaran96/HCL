import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';

function AddUser() {
  const [form, setForm] = useState({ name: '', age: '', city: '', state: '', pincode: '' });
  const [errors, setErrors] = useState({});
  const [submitting, setSubmitting] = useState(false);
  const [apiError, setApiError] = useState(null);
  const navigate = useNavigate();

  const validate = () => {
    const errs = {};
    if (!form.name || form.name.length < 2 || form.name.length > 100) errs.name = 'Name must be 2-100 characters';
    if (!form.age || isNaN(form.age) || form.age < 0 || form.age > 120) errs.age = 'Age must be 0-120';
    if (!form.city) errs.city = 'City is required';
    if (!form.state) errs.state = 'State is required';
    if (!form.pincode || form.pincode.length < 4 || form.pincode.length > 10) errs.pincode = 'Pincode must be 4-10 characters';
    return errs;
  };

  const handleChange = e => {
    setForm({ ...form, [e.target.name]: e.target.value });
  };

  const handleSubmit = e => {
    e.preventDefault();
    const errs = validate();
    if (Object.keys(errs).length > 0) {
      setErrors(errs);
      return;
    }
    setErrors({});
    setSubmitting(true);
    setApiError(null);

    fetch('/api/users', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        name: form.name,
        age: Number(form.age),
        city: form.city,
        state: form.state,
        pincode: form.pincode
      })
    })
      .then(res => {
        if (!res.ok) throw new Error('Failed to add user');
        return res.json();
      })
      .then(() => {
        alert('User added successfully');
        navigate('/list');
      })
      .catch(err => {
        setApiError(err.message);
        setSubmitting(false);
      });
  };

  return (
    <form onSubmit={handleSubmit} style={{ maxWidth: 400, margin: 'auto' }}>
      <div>
        <label>Name:</label><br />
        <input name="name" value={form.name} onChange={handleChange} />
        {errors.name && <div style={{ color: 'red' }}>{errors.name}</div>}
      </div>
      <div>
        <label>Age:</label><br />
        <input name="age" type="number" value={form.age} onChange={handleChange} />
        {errors.age && <div style={{ color: 'red' }}>{errors.age}</div>}
      </div>
      <div>
        <label>City:</label><br />
        <input name="city" value={form.city} onChange={handleChange} />
        {errors.city && <div style={{ color: 'red' }}>{errors.city}</div>}
      </div>
      <div>
        <label>State:</label><br />
        <input name="state" value={form.state} onChange={handleChange} />
        {errors.state && <div style={{ color: 'red' }}>{errors.state}</div>}
      </div>
      <div>
        <label>Pincode:</label><br />
        <input name="pincode" value={form.pincode} onChange={handleChange} />
        {errors.pincode && <div style={{ color: 'red' }}>{errors.pincode}</div>}
      </div>
      {apiError && <div style={{ color: 'red' }}>{apiError}</div>}
      <button type="submit" disabled={submitting}>{submitting ? 'Submitting...' : 'Add User'}</button>
    </form>
  );
}

export default AddUser;
